\section{Cyclostationary analysis}

A signal with the property $x(t) = x(t + T)\ \forall\, t,\ T>0$ is called \textit{strictly periodic} (or \textit{periodic pointwise}).
A composition of periodic signals $h(t) = \sum_{i=0}^N x_i(t)$ is also periodic $\iff \tfrac{T_i}{T_j} \in \mathbb{Q}\ \forall\, i,j$ (i.e., the periods are commensurable).
The period of $h(t)$ can therefore be described as
\[
T = \mathrm{lcm}(T_0, T_1, \dots, T_N).
\]

\noindent\textbf{Finite field computation of period.} For FPGA implementation, the LCM computation can be performed efficiently using \emph{finite field factorization} over a Galois field $\mathrm{GF}(p^k)$. Represent each period $T_i$ as a polynomial in $\mathrm{GF}(p)[x]$, where $p$ is chosen such that all period numerators and denominators (when expressed as rationals $T_i = n_i/d_i$) can be embedded in the field. The period LCM then reduces to computing the \emph{polynomial LCM} via Berlekamp's algorithm~\cite{berlekamp1970factoring,gathen2013modern}:
\begin{enumerate}
\item Factor each period polynomial: $T_i(x) = \prod_{j} f_{i,j}(x)^{e_{i,j}}$ over $\mathrm{GF}(p)$ using Berlekamp factorization (runtime $O(d^3 + d^2 \log p)$ for degree-$d$ polynomials).
\item Compute LCM via exponent maximization: $\mathrm{lcm}(T_0, \ldots, T_N) = \prod_{f \text{ irreducible}} f(x)^{\max_i e_{i,f}}$.
\end{enumerate}

\noindent\textbf{Example:} Consider three sinusoids with periods $T_0 = 3$, $T_1 = 4$, $T_2 = 6$. In $\mathrm{GF}(7)$ (choosing prime $p=7 > \max\{T_i\}$):
\begin{align}
T_0 &= 3 = 3 \quad \text{(irreducible in } \mathrm{GF}(7)\text{)}, \\
T_1 &= 4 = 2^2 \quad \text{(} 2 \text{ irreducible)}, \\
T_2 &= 6 = 2 \cdot 3 \quad \text{(factorization over } \mathrm{GF}(7)\text{)}.
\end{align}
Thus $\mathrm{lcm}(3, 4, 6) = 2^2 \cdot 3 = 12$.

For the \emph{quadratic case}, suppose periods arise from solving a characteristic equation $\lambda^2 + a\lambda + b = 0$ over $\mathrm{GF}(p)$ (common in second-order filters). Berlekamp's algorithm factors this as:
\[
\lambda^2 + a\lambda + b = (\lambda - r_1)(\lambda - r_2) \quad \text{over } \mathrm{GF}(p),
\]
where $r_1, r_2 \in \mathrm{GF}(p)$ are the roots (if they exist). The factorization determines the minimal periods $T_1 = \text{ord}(r_1)$, $T_2 = \text{ord}(r_2)$ (the multiplicative orders of $r_1, r_2$ in $\mathrm{GF}(p)^*$), and the composite period is $T = \mathrm{lcm}(T_1, T_2)$.

\noindent\textbf{Hardware advantage:} Berlekamp factorization uses only addition, multiplication, and GCD operations in $\mathrm{GF}(p)$—all implementable with fixed-width arithmetic on FPGA. This avoids arbitrary-precision integer LCM, reducing DSP slice usage for cyclostationary period detection.

\medskip
\noindent A \textit{wide-sense cyclostationary} (WSC) signal is defined by
\[
\mathbb{E}[x(t)] = \mathbb{E}[x(t+T)], \quad R_x(t,\tau) = R_x(t+T,\tau),
\]
where $R_x(t,\tau) = \mathbb{E}[x(t)x^*(t+\tau)]$.
That is, both the mean $\bar{x}(t)$ and the autocorrelation function $R_x(t,\tau)$ are periodic in $t$.
More generally, a signal is cyclostationary \textit{iff} there exists some $n$th-order nonlinear transformation of $x(t)$ that produces a finite additive sinusoidal component~\cite{}.
\medskip

\noindent The definition of the \textit{spectral parameter}~\cite{} is
\[
\hat{x}(\alpha) \;=\; \lim_{T \to \infty} \frac{1}{T}
\int_{-T/2}^{T/2} x(t)\,e^{-j2\pi \alpha t}\,dt.
\]

For $x(t) = a\cos(2\pi \beta t + \theta)$:
\[
x(t) = \tfrac{a}{2}\big(e^{j(2\pi \beta t + \theta)} + e^{-j(2\pi \beta t + \theta)}\big).
\]

Substituting into the definition:
\[
\hat{x}(\alpha) = \lim_{T \to \infty} \frac{a}{2T}
\left[
e^{j\theta}\int_{-T/2}^{T/2} e^{j2\pi(\beta-\alpha)t}\,dt \;+\;
e^{-j\theta}\int_{-T/2}^{T/2} e^{-j2\pi(\beta+\alpha)t}\,dt
\right].
\]

Recalling that
\[
\int_{-T/2}^{T/2} e^{j2\pi f t}\,dt = \frac{\sin(\pi f T)}{\pi f},
\]

\[
\implies \hat{x}(\alpha) = \frac{a}{2} \lim_{T \to \infty} \frac{1}{T}\Bigg[
e^{j\theta}\frac{\sin(\pi(\beta-\alpha)T)}{\pi(\beta-\alpha)}
+ e^{-j\theta}\frac{\sin(\pi(\beta+\alpha)T)}{\pi(\beta+\alpha)}
\Bigg].
\]

Equivalently,
\[
\hat{x}(\alpha) = \frac{a}{2}\lim_{T \to \infty}\left[e^{j\theta}\,\mathrm{sinc}_\pi((\beta-\alpha)T)
+ e^{-j\theta}\,\mathrm{sinc}_\pi((\beta+\alpha)T)\right],
\]

Finally, by inspection we obtain:
\[
\hat{x}(\alpha) =
\begin{cases}
	\dfrac{a}{2}e^{j\theta}, & \alpha = \beta,\\[6pt]
	\dfrac{a}{2}e^{-j\theta}, & \alpha = -\beta,\\[6pt]
	0, & \text{otherwise}.
\end{cases}
\]

This demonstrates that the time-averaged inner product survives only when the test frequency $\alpha$ matches a sinusoidal component $\beta$ of the signal.
\medskip

\noindent Furthermore, we can interpret $\hat{x}(\alpha)$ as arising from the filtered projector framework developed in Chapter~3. Recall from Section~3.2 that for an analytic kernel $H_a(\omega)$, the filtered projector is evaluated via contour integration:
\[
\mathbb{P}_N[\text{sinc} * h](z) = \frac{1}{2\pi i}\oint \frac{S_{\text{even}}(\omega) \cdot H_{a,\text{even}}(\omega)}{\omega^2+z^2} d\omega + \frac{z}{2\pi}\oint \frac{S_{\text{odd}}(\omega) \cdot H_{a,\text{odd}}(\omega)}{\omega^2+z^2} d\omega.
\]
The spectral parameter $\hat{x}(\alpha)$ corresponds to choosing $H_a(\omega) = e^{j\omega T/2}\mathrm{sinc}_\pi(\epsilon\omega)$ (an ideal bandpass kernel centered at frequency $\alpha$), with the limit $\epsilon \to 0$ isolating the spectral component. Evaluating via residues at the poles $\omega = \pm iz$ yields:
\[
\hat{x}(\alpha) = \lim_{T \to \infty} \frac{1}{T} \int_{-T/2}^{T/2} x(t) e^{-j2\pi\alpha t} \, dt = \text{Res}_{\omega = i\alpha} \left[\frac{X(\omega) \cdot H_a(\omega)}{\omega^2 + \alpha^2}\right],
\]
connecting cyclostationary spectral analysis to the contour-based framework. This interpretation enables runtime computation of spectral parameters using the same residue evaluation infrastructure developed for adaptive filter quantization (Chapter~3, Section~3.2), unifying the signal analysis (Chapter~4) and resource optimization (Chapter~3) pipelines.

\subsection{Relation between Strict Periodicity and Cyclostationarity}

Every strictly periodic signal is trivially wide-sense cyclostationary.
Indeed, if $x(t) = x(t+T)$, then
\[
\mathbb{E}[x(t)] = \mathbb{E}[x(t+T)], \quad
R_x(t,\tau) = \mathbb{E}[x(t)x^*(t+\tau)] = \mathbb{E}[x(t+T)x^*(t+T+\tau)] = R_x(t+T,\tau).
\]
Thus:
\[
x(t)\ \text{periodic} \;\implies\; x(t)\ \text{WSC (cyclostationary)}.
\]

The converse does not hold: cyclostationary signals need not be strictly periodic pointwise.
For example, an amplitude-modulated process $x(t) = m(t)\cos(2\pi f_c t)$ with random data $m(t)$ can be cyclostationary with period $1/f_c$, while not being pointwise periodic.

\subsection{Group-Theoretic Interpretation}
The set of transformations $T$ can be understood as the action of a \emph{symmetry group} $G$ on the domain $A$. In the sine example, the relevant group is the \emph{dihedral group} $D_4$, which consists of rotations and reflections of a square \cite{armstrong1988groups, gallian2021contemporary}.
Each $\tau \in T$ corresponds to an element of $D_4$ acting on the domain of $\sin(x)$, with reflections accounting for sign changes and rotations corresponding to translations by $\pi/2$.

Thus, the concept of foldability is equivalent to requiring that the function $f$ is invariant under the action of a group $G$, up to sign or phase factors. The \emph{fundamental domain} $A_0$ is then a representative region of $A$ under this group action, and folding reduces storage by exploiting orbit representatives of $G$.

In group-theoretic language, the prior example is invariant under the group action up to a sign factor, i.e.,
\[
\sin(x) = \epsilon(\tau) \, \sin(\tau(x)), \quad \epsilon(\tau) \in \{+1, -1\}, \ \tau \in G.
\]
The fundamental domain $A_0$ corresponds to a set of orbit representatives of $G$ acting on $A$, and the recovery method reconstructs the full signal from this reduced domain.

\subsection{Relationship between Argument Reduction and Cyclostationarity}

\begin{lemma}[Reducible function definition]\label{eq:reducible_function}
A function $f: A \to B$ is \emph{reducible} if and only if there exist a subset $A_0 \subseteq A$ and a set of transformations $T = \{ \tau_i : A \to A \}_{i \in I}$ such that
\[
\forall x \in A, \, \exists \, \tau \in T \text{ with } \tau(x) \in A_0
\quad \text{and} \quad f(x) = f \circ \tau(x).
\]
\end{lemma}

Where $A_0$ is called the \textit{fundamental domain} \& $T$, the set of transformations mapping from $A_0$ to $A$, the \textit{recovery method}.
\medskip

\noindent Argument-reducible functions are naturally related to cyclostationarity. A cyclostationary signal $x(t)$ has statistics that are periodic with period $T$.
For an argument-reducible function like $\sin(x)$, the fundamental domain $A_0$ can be interpreted as a segment over which the first- and second-order statistics repeat under the action of $G$. In other words, argument reduction exploits the same periodic structure that cyclostationary analysis detects: the mean and autocorrelation within $A_0$ are sufficient to reconstruct the signal statistics over the entire domain.
\medskip

\noindent Thus, argument reducibility can be seen as a deterministic, structural analogue of cyclostationarity: instead of averaging over time to detect periodic statistics, we use group-theoretic transformations to reconstruct the full signal from a reduced representative domain.
\bigskip

\noindent Let \( f: A \to B \) be a reducible function with fundamental domain \( A_0 \subseteq A \) and recovery transformations \( T = \{ \tau_i : A \to A \}_{i \in I} \) such that
\[
\forall x \in A, \exists \tau \in T \text{ with } \tau(x) \in A_0 \quad \text{and} \quad f(x) = f \circ \tau(x).
\]
This implies that the function \( f \) is invariant under the group action of \( G \) generated by \( T \), and the entire domain \( A \) can be reconstructed from \( A_0 \) using the transformations in \( T \).
\medskip

\noindent Now, consider the autocorrelation function of \( f \):
\[
R_f(x, \Delta) = \mathbb{E}[f(x) f^*(x + \Delta)].
\]
Since \( f(x) = f(\tau(x)) \) for some \( \tau \in T \), we have
\[
R_f(x, \Delta) = \mathbb{E}[f(\tau(x)) f^*(\tau(x + \Delta))].
\]
Because \( \tau(x), \tau(x + \Delta) \in A_0 \) (up to another element of the group), all evaluations of \( R_f(x, \Delta) \) are determined entirely by the behavior on \( A_0 \). Therefore,
\[
R_f(x, \Delta) = R_f(\tau(x), \Delta), \quad \forall x \in A, \ \tau \in T, \ \tau(x) \in A_0.
\]
Thus, the fundamental domain \( A_0 \) captures all statistical information of the function \( f \), analogous to how the fundamental domain in cyclostationary signal processing captures the periodic statistical properties of a signal.

\subsubsection{Projector-Based Detection of Group Structure in Wavelet Synthesis}

When synthesizing waveforms using an orthonormal basis such as the Shannon wavelet mother function $\psi(t) = \text{sinc}(t)$ (Chapter~3, Section~3.4), the \emph{filtered projector} framework from Chapter~3, Section~3.2 provides a direct mechanism to detect both the group action $G$ and the recovery transformations $T$.

\paragraph{Synthesis via Orthonormal Expansion}

Suppose we synthesize a target waveform $f(t)$ using the Shannon wavelet basis:
\[
f(t) = \sum_{j \in \mathbb{Z}} \sum_{k \in \mathbb{Z}} c_{j,k} \psi_{j,k}(t), \quad \psi_{j,k}(t) = 2^{j/2} \psi(2^j t - k),
\]
where coefficients $c_{j,k} = \langle f, \psi_{j,k} \rangle$ are computed via the filtered projector:
\[
c_{j,k} = \int_{-\infty}^\infty f(t) \overline{\psi_{j,k}(t)} \, dt = \mathbb{P}_{\psi_{j,k}}[f],
\]
with $\mathbb{P}_{\psi_{j,k}}$ denoting projection onto the subspace spanned by $\psi_{j,k}$. For functions with argument reduction symmetry, these coefficients exhibit \emph{algebraic structure} that reveals the underlying group $G$.

\paragraph{Group Action Detection via Coefficient Symmetries}

\noindent\textbf{Key observation:} If $f(t)$ is invariant under transformations $\tau \in T$ (up to phase factors), then the wavelet coefficients satisfy:
\[
c_{j,k} = \epsilon(\tau) \cdot c_{j, \tau_k(k)}, \quad \epsilon(\tau) \in \{\pm 1, e^{i\theta}\},
\]
where $\tau_k(k)$ denotes the action of $\tau$ on the discrete index $k$ at scale $j$. This coefficient symmetry is \emph{detectable} by analyzing the filtered projector integrals.

\noindent\textbf{Example: Sine function.} For $f(t) = \sin(2\pi t)$ with period $T=1$, the symmetry group is $D_4$ (dihedral group). The recovery transformations are:
\begin{align}
\tau_1(t) &= t, \quad &&\text{(identity)}, \\
\tau_2(t) &= -t, \quad &&\text{(reflection, introduces sign flip)}, \\
\tau_3(t) &= \frac{1}{2} - t, \quad &&\text{(quarter-period shift + reflection)}, \\
\tau_4(t) &= t + \frac{1}{2}. \quad &&\text{(half-period translation)}.
\end{align}

For Shannon wavelets at scale $j$ (covering frequency band $[2^{j-1}\pi, 2^j\pi]$), the projector integrals become:
\[
c_{j,k} = \int_{-\infty}^\infty \sin(2\pi t) \cdot 2^{j/2} \text{sinc}(2^j t - k) \, dt.
\]
Using the contour integration method from Chapter~3, Section~3.2, we evaluate this via residue calculus. The key insight is that $\sin(2\pi t)$ has Fourier transform $S(\omega) = \frac{i}{2}[\delta(\omega - 2\pi) - \delta(\omega + 2\pi)]$, so:
\[
c_{j,k} = 2^{j/2} \int_{-\infty}^\infty S(\omega) \widehat{\psi}(2^{-j}\omega) e^{i\omega k / 2^j} \, \frac{d\omega}{2\pi},
\]
where $\widehat{\psi}(\omega) = \mathbb{1}_{[-\pi,\pi]}(\omega)$ (brick-wall filter). The integral is nonzero only when $2\pi \in [2^{j-1}\pi, 2^j\pi]$, i.e., when $j=1$. For $j=1$:
\[
c_{1,k} = \sqrt{2} \int_{-2\pi}^{2\pi} \frac{i}{2}[\delta(\omega - 2\pi) - \delta(\omega + 2\pi)] e^{i\omega k / 2} \, \frac{d\omega}{2\pi} = \frac{i\sqrt{2}}{4\pi}[e^{i\pi k} - e^{-i\pi k}] = \frac{\sqrt{2}}{2\pi}\sin(\pi k).
\]

\noindent\textbf{Detected symmetry:} The coefficients satisfy $c_{1,k} = -c_{1,-k}$ (antisymmetry under reflection $k \to -k$) and $c_{1,k+2} = -c_{1,k}$ (half-period shift), directly encoding the $D_4$ group action. The \emph{fundamental domain} is $k \in \{0, 1\}$ (one quarter-period), from which all coefficients can be reconstructed via $\epsilon(\tau) \in \{+1, -1\}$.

\paragraph{Algorithmic Recovery Method Identification}

Given synthesized coefficients $\{c_{j,k}\}$ from the filtered projector, we algorithmically detect the group $G$ and recovery method $T$:

\begin{enumerate}
\item \textbf{Correlation matrix construction:} For each scale $j$, compute the \emph{coefficient autocorrelation}:
\[
\rho_j(\Delta k) = \frac{\sum_k c_{j,k} c_{j,k+\Delta k}^*}{\sum_k |c_{j,k}|^2}.
\]
This is the discrete analogue of $R_f(x, \Delta)$ from Section~4.1.3. Peaks in $|\rho_j(\Delta k)|$ reveal candidate periodicities.

\item \textbf{Galois field factorization for period detection:} The dominant periodicity $\Delta k_0$ (where $|\rho_j(\Delta k_0)| \approx 1$) corresponds to the fundamental period $T$ in index space. Using the finite field LCM method from Section~4.1 (Berlekamp factorization), decompose $\Delta k_0$ into irreducible factors over $\mathrm{GF}(p)$. Each factor corresponds to a generator of the cyclic subgroup of $G$.

\item \textbf{Phase factor extraction:} Compute the coefficient ratio:
\[
\epsilon_j(\Delta k) = \frac{c_{j,k+\Delta k}}{c_{j,k}}, \quad \text{(for } c_{j,k} \neq 0\text{)}.
\]
If $|\epsilon_j(\Delta k)| \approx 1$ for all $k$, the transformation $\tau: k \to k + \Delta k$ is in $T$ with phase factor $\epsilon(\tau) = \arg(\epsilon_j(\Delta k))$. For real-valued functions, $\epsilon(\tau) \in \{+1, -1\}$ (parity symmetry).

\item \textbf{Fundamental domain extraction:} The fundamental domain $A_0$ in index space is the minimal set $\{k_0, k_1, \ldots, k_{N-1}\}$ such that all other coefficients $c_{j,k}$ can be expressed as:
\[
c_{j,k} = \epsilon(\tau_{k \to k'}) \cdot c_{j,k'}, \quad k' \in A_0, \quad \tau_{k \to k'} \in T.
\]
This is solved via \emph{orbit decomposition} under the detected group $G$ (computed using the Schreier-Sims algorithm~\cite{seress2003permutation} for permutation groups).
\end{enumerate}

\paragraph{Connection to Filtered Projector Residue Structure}

The filtered projector from Chapter~3, Section~3.2, evaluates coefficients via contour integrals:
\[
c_{j,k} = \frac{1}{2\pi i} \oint_{\mathcal{C}} \frac{S(\omega) \widehat{\psi}(2^{-j}\omega) e^{i\omega k / 2^j}}{\omega^2 - \omega_0^2} \, d\omega,
\]
where $S(\omega) = \mathcal{F}[f(t)]$ and $\mathcal{C}$ is a contour enclosing the singularities of $S(\omega)$. For argument-reducible functions, the singularities of $S(\omega)$ exhibit \emph{symmetry} under the dual group $\widehat{G}$ (Pontryagin duality). Specifically:
\begin{itemize}
\item \textbf{Reflection symmetry} ($\tau(t) = -t$) $\implies$ $S(-\omega) = S(\omega)$ (even spectrum) or $S(-\omega) = -S(\omega)$ (odd spectrum).
\item \textbf{Translation symmetry} ($\tau(t) = t + T$) $\implies$ $S(\omega)$ has support only at integer multiples of $2\pi/T$ (comb spectrum).
\end{itemize}

The residue calculus then yields coefficient relations. For the sine example, $S(\omega) = \frac{i}{2}[\delta(\omega - 2\pi) - \delta(\omega + 2\pi)]$ has \emph{antisymmetric poles}, forcing $c_{j,k} = -c_{j,-k}$ (detected in step 3 above). More generally, the \emph{residue signature}:
\[
\text{Res}_{\omega = \omega_n} \left[ \frac{S(\omega) e^{i\omega k}}{\omega^2 - \omega_0^2} \right] \propto e^{i\omega_n k},
\]
encodes the group generators via the phase factors $e^{i\omega_n T} = \epsilon(\tau)$ where $\tau$ is a translation by period $T$.

\paragraph{Practical Implementation for Wavelet-Based Synthesis}

In Algorithm~4.1 (Section~4.2.5), the offline training phase computes Shannon wavelet coefficients $c_{j,k}$ for representative signals. We now augment this with \emph{group action detection}:

\begin{enumerate}
\item[\textbf{1a.}] Compute coefficient autocorrelations $\rho_j(\Delta k)$ per scale (O($N \log N$) via FFT).
\item[\textbf{1b.}] Apply Berlekamp factorization (Section~4.1) to detected periods $\Delta k_0$ over $\mathrm{GF}(p)$.
\item[\textbf{1c.}] Extract phase factors $\epsilon(\tau)$ and construct group table for $G$.
\item[\textbf{1d.}] Identify fundamental domain $A_0$ via orbit decomposition.
\item[\textbf{1e.}] Store only coefficients $\{c_{j,k} \mid k \in A_0\}$ and recovery table $\{(\tau, \epsilon(\tau)) \mid \tau \in T\}$.
\end{enumerate}

At synthesis time (runtime execution), reconstruct full coefficient set via:
\[
c_{j,k} = \epsilon(\tau_{k'}) \cdot c_{j,k'}, \quad k' = \arg\min_{k'' \in A_0} |k - \tau_{k''}^{-1}(k'')|,
\]
reducing memory by factor $|G|$ (e.g., $4\times$ for $D_4$ symmetry in sine/cosine). This directly implements the argument reduction framework from Chapter~5, Section~5.1, within the Koopman-MPC resource optimization pipeline.

\noindent\textbf{Hardware advantage:} On FPGA, the group table $\{(\tau, \epsilon(\tau))\}$ is implemented as a small BRAM lookup (typically $< 1$ Kb for common symmetries), while coefficient storage in main BRAM is reduced by factor $|G|$. The Berlekamp factorization (Section~4.1) runs in hardware using Galois field arithmetic, enabling real-time period detection for adaptive synthesis.

\section{Koopman Operator Framework for DSP Resource Prediction}

The cyclostationary analysis developed in Section~4.1 provides a statistical characterization of periodic workloads, but does not directly address the \emph{dynamics} of how DSP modules transform signals over time. To bridge this gap, we employ the \emph{Koopman operator} framework~\cite{Koopman1931,Budisic2012,Mauroy2020Book}, which represents nonlinear dynamical systems as linear infinite-dimensional operators on spaces of observable functions.

\subsection{Motivation: From Signal Analysis to Resource Prediction}

Consider a modular audio signal chain comprising filters, nonlinear processors (distortion, compression), and time-varying effects (chorus, phaser). Each module $\mathcal{M}_i$ can be viewed as a dynamical system:
\[
\mathbf{x}_{k+1} = \mathcal{F}_i(\mathbf{x}_k, u_k), \quad y_k = \mathcal{H}_i(\mathbf{x}_k),
\]
where $\mathbf{x}_k \in \mathbb{R}^n$ is the internal state (e.g., filter delay line, envelope detector state), $u_k$ is the input signal, and $y_k$ is the output. For nonlinear effects like tube saturation or dynamic range compression, $\mathcal{F}_i$ is inherently nonlinear, making traditional linear systems analysis inadequate.

The Koopman operator~\cite{Mezic2005} provides a linearization by lifting the nonlinear dynamics to an infinite-dimensional space of observables. This enables:
\begin{enumerate}
\item \textbf{Linear prediction} of future signal characteristics from past observations~\cite{Williams2015,Proctor2016}, crucial for anticipatory resource allocation.
\item \textbf{Spectral decomposition} into Koopman modes, which directly correspond to frequency-band activity—precisely the structure exploited by the Shannon wavelet basis (Chapter~3, Section~3.4).
\item \textbf{Data-driven identification} via Dynamic Mode Decomposition (DMD)~\cite{Schmid2010}, avoiding the need for explicit nonlinear system models.
\end{enumerate}

Recent work demonstrates the efficacy of Koopman methods for audio applications: Huhtala et al.~\cite{Huhtala2024} model nonlinear guitar amplifier dynamics, while Díaz and Schlecht~\cite{DiazSchlecht2024} apply extended Koopman DMD to capture modal interactions in reverberant systems. We extend these ideas to \emph{resource management}, using Koopman-linearized dynamics to predict FPGA workload demands.

\subsection{Koopman Operator Theory}

Let $\mathcal{F}: \mathcal{X} \to \mathcal{X}$ be a dynamical system on state space $\mathcal{X}$. The Koopman operator $\mathcal{K}: \mathcal{G} \to \mathcal{G}$ acts on the space of observables $\mathcal{G} = \{g: \mathcal{X} \to \mathbb{C}\}$ by composition:
\[
(\mathcal{K} g)(\mathbf{x}) = g(\mathcal{F}(\mathbf{x})).
\]

\noindent\textbf{Key property:} While $\mathcal{F}$ may be nonlinear, $\mathcal{K}$ is \emph{linear} by construction~\cite{Koopman1931}:
\[
\mathcal{K}(\alpha g_1 + \beta g_2) = \alpha \mathcal{K} g_1 + \beta \mathcal{K} g_2, \quad \forall \alpha, \beta \in \mathbb{C}, \ g_1, g_2 \in \mathcal{G}.
\]

For continuous-time systems $\dot{\mathbf{x}} = \mathbf{f}(\mathbf{x})$, the infinitesimal generator $\mathcal{L}$ of the Koopman semigroup satisfies~\cite{Mezic2005}:
\[
\mathcal{L} g = \nabla g \cdot \mathbf{f}(\mathbf{x}).
\]

\subsection{Connection to Shannon Wavelet Decomposition}

\noindent\textbf{Clarification:} The Koopman operator framework developed here is \emph{independent} of the cyclostationary analysis in Section~4.1. While cyclostationarity characterizes \emph{statistical periodicity} in signal ensembles, the Koopman approach models \emph{deterministic dynamics} of individual DSP modules. Both perspectives contribute to resource prediction: cyclostationary analysis detects periodic workload patterns, while Koopman operators predict transient responses to signal changes.

The Koopman spectral decomposition exhibits a profound connection to the Shannon wavelet framework developed in Chapter~3. Suppose the Koopman operator $\mathcal{K}$ admits a discrete spectrum with eigenfunctions $\{\varphi_j\}$ and eigenvalues $\{\lambda_j\}$:
\[
\mathcal{K} \varphi_j = \lambda_j \varphi_j, \quad \lambda_j = e^{j\omega_j \Delta t},
\]
where $\omega_j$ are the Koopman frequencies. An observable $g(\mathbf{x})$ can be expanded:
\[
g(\mathbf{x}) = \sum_{j=0}^\infty v_j \varphi_j(\mathbf{x}), \quad v_j = \langle g, \varphi_j \rangle,
\]
yielding the evolution:
\[
g(\mathcal{F}^k(\mathbf{x})) = \mathcal{K}^k g(\mathbf{x}) = \sum_{j=0}^\infty v_j \lambda_j^k \varphi_j(\mathbf{x}) = \sum_{j=0}^\infty v_j e^{jk\omega_j \Delta t} \varphi_j(\mathbf{x}).
\]

\noindent\textbf{Wavelet interpretation.} Each Koopman mode oscillates at frequency $\omega_j$, analogous to the Shannon wavelet basis functions $\psi_{j,k}(t)$ which isolate octave bands $[2^j\pi, 2^{j+1}\pi]$ (Section~3.4.1). The Koopman eigenfunction $\varphi_j(\mathbf{x})$ encodes \emph{which states} contribute to frequency $\omega_j$, while the Shannon wavelet coefficient $\langle x(t), \psi_{j,k}(t) \rangle$ measures \emph{signal energy} in that band.

This duality enables a unified framework:
\begin{enumerate}
\item \textbf{Koopman modes} $\to$ predict which frequency bands will be active based on system state $\mathbf{x}_k$.
\item \textbf{Shannon wavelets} $\to$ allocate bits per band (Chapter~3, Section~3.4.5) based on predicted energy $|v_j|^2$.
\item \textbf{Filtered projectors} (Chapter~3, Section~3.2) $\to$ quantize coefficients for kernels $H(\omega)$ derived from Koopman eigenfunctions.
\end{enumerate}

\subsubsection{Dyadic Quantization and Integer Arithmetic Optimization}

A key design parameter in the Shannon wavelet mother function is whether to impose \emph{dyadic quantization} on frequency band edges. Recall from Chapter~3 that the Shannon wavelet $\psi(t) = \text{sinc}(t)$ has ideal brick-wall spectrum support on $[-\pi, \pi]$. The multiresolution decomposition naturally yields octave bands $[\pi 2^j, \pi 2^{j+1}]$ for scales $j \in \mathbb{Z}$.

When implementing this on FPGA, we can leverage the dyadic structure:
\begin{itemize}
\item \textbf{Dyadic mode enabled:} Band edges at powers-of-two frequencies $\{2^j \pi\}$ allow multiply-accumulate operations to be replaced with bit shifts: $x \cdot 2^j \to x \ll j$. The filtered projector kernels $H(\omega)$ from Chapter~3, Section~3.2 become particularly efficient when $\omega = 2^j \alpha$ for some base frequency $\alpha$, as the contour integral residues simplify~\cite{Unser2000}.

\item \textbf{Dyadic mode disabled:} Arbitrary band partitions $[\omega_j, \omega_{j+1}]$ with non-power-of-two edges require full floating-point or high-precision fixed-point multipliers. This is necessary when Koopman spectral analysis reveals that the dominant frequencies $\{\omega_j\}$ do not align with dyadic grid—common in modulated effects (chorus, flanger) where pitch-shifting introduces irrational frequency ratios.
\end{itemize}

\noindent\textbf{Adaptive strategy:} The Koopman operator $\mathbf{K}$ learned via EDMD (Section~4.2.4) provides the eigenfrequencies $\{\omega_j\}$. If the dominant Koopman modes satisfy $|\omega_j - 2^{n_j}\pi| < \epsilon_{\text{snap}}$ for integer exponents $\{n_j\}$, we \emph{enable dyadic quantization} and snap band edges to the nearest power-of-two. The Shannon wavelet transform then reduces to integer arithmetic:
\begin{align}
c_{j,k} &= \langle x(t), \psi_{j,k}(t) \rangle = \int_{-\infty}^\infty x(t) \, 2^{j/2} \psi(2^j t - k) \, dt, \\
&\approx 2^{j/2} \sum_{n} x[n] \, \psi_{\text{discrete}}[2^j n - k], \quad \text{(bit-shift implementation)}.
\end{align}
Conversely, if Koopman modes are incommensurate with dyadic grid, we revert to non-dyadic partitions and accept the higher computational cost for accuracy.

\noindent\textbf{Wave synthesis implications:} For waveform generation (Chapter~5), dyadic Shannon wavelets enable extremely fast additive synthesis. Given target spectrum $S(\omega)$ supported on $[2^j \pi, 2^{j+1}\pi]$, the synthesis becomes:
\[
x(t) = \sum_{j} \sum_{k} c_{j,k} \psi_{j,k}(t), \quad c_{j,k} = 2^{-j/2} \int_{2^j \pi}^{2^{j+1}\pi} S(\omega) e^{j\omega k / 2^j} \, d\omega.
\]
With dyadic quantization, all frequency-domain multiplications $S(\omega) \cdot \omega$ reduce to shifts, and inverse FFT lengths are powers-of-two (enabling radix-2 Cooley-Tukey algorithms~\cite{Muller2006}). The Koopman framework predicts \emph{which bands} $j$ will require synthesis, allowing us to:
\begin{enumerate}
\item Preallocate BRAM for only active dyadic scales.
\item Configure CORDIC units~\cite{Meher2008} with bit-shift mode for $2^j$ multipliers.
\item Skip entire octaves where Koopman mode amplitudes $|v_j| < \epsilon_{\text{thresh}}$.
\end{enumerate}

This creates a feedback loop: Koopman analysis $\to$ identifies dominant frequencies $\to$ determines dyadic/non-dyadic mode $\to$ optimizes Shannon wavelet implementation $\to$ measured FPGA resource usage $\to$ refines Koopman observables $\psi_{\text{DSP}}, \psi_{\text{LUT}}$.
\begin{equation}
\boxed{
\text{Koopman } \omega_j \xrightarrow{\text{dyadic test}} \text{Mode selection} \xrightarrow{\text{synthesis}} \text{FPGA utilization} \xrightarrow{\text{measurement}} \text{Update } \{\alpha_j, \beta_j\}
}
\label{eq:dyadic_feedback_loop}
\end{equation}

\subsection{Finite-Dimensional Approximation via Extended DMD}

Practical implementation requires finite-dimensional truncation. The Extended Dynamic Mode Decomposition (EDMD)~\cite{Williams2015} constructs an approximate Koopman operator $\mathbf{K} \in \mathbb{C}^{m \times m}$ from snapshot pairs $\{(\mathbf{x}_k, \mathbf{x}_{k+1})\}_{k=1}^{N}$ and a dictionary of observables $\{\psi_i(\mathbf{x})\}_{i=1}^{m}$:
\[
\mathbf{\Psi}(\mathbf{x}) = \begin{bmatrix} \psi_1(\mathbf{x}) \\ \vdots \\ \psi_m(\mathbf{x}) \end{bmatrix}, \quad
\mathbf{K} = \mathbf{\Psi}' (\mathbf{\Psi})^\dagger,
\]
where $\mathbf{\Psi} = [\mathbf{\Psi}(\mathbf{x}_1), \ldots, \mathbf{\Psi}(\mathbf{x}_N)]$, $\mathbf{\Psi}' = [\mathbf{\Psi}(\mathbf{x}_2), \ldots, \mathbf{\Psi}(\mathbf{x}_{N+1})]$, and $(\cdot)^\dagger$ denotes the Moore-Penrose pseudoinverse.

For audio DSP, recent work~\cite{Huhtala2024} demonstrates that polynomial observables $\psi_i(\mathbf{x}) = x_1^{a_1} \cdots x_n^{a_n}$ with $a_1 + \cdots + a_n \leq d$ (degree-$d$ monomials) effectively capture nonlinear audio effects with $m = O(n^d)$ observables. Brunton et al.~\cite{Brunton2016} show that for control applications, carefully chosen observables yield Koopman invariant subspaces of dimension $m \ll \infty$, enabling real-time computation.

\subsection{Resource Prediction via Koopman-MPC Integration}

The Koopman linearization enables Model Predictive Control (MPC) for resource allocation~\cite{Korda2018}. Given the finite-dimensional Koopman operator $\mathbf{K}$, the predicted state evolution is:
\[
\mathbf{\Psi}_{k+h|k} = \mathbf{K}^h \mathbf{\Psi}_k,
\]
where $h$ is the prediction horizon. We define resource demand observables:
\begin{align}
\psi_{\text{DSP}}(\mathbf{x}) &= \sum_{j} \alpha_j |\langle x(t), \psi_{j,\cdot}(t) \rangle|^2, \quad \text{(DSP slice usage)},\\
\psi_{\text{LUT}}(\mathbf{x}) &= \sum_{j} \beta_j N_j, \quad \text{(LUT count for band $j$)},\\
\psi_{\text{BRAM}}(\mathbf{x}) &= \sum_{j} \gamma_j M_j, \quad \text{(BRAM for delay lines)},
\end{align}
where $\alpha_j, \beta_j, \gamma_j$ are learned coefficients mapping Shannon wavelet energies to hardware primitives.

\subsubsection{Sensitivity-Guided Observable Selection for Resource Tradeoffs}

The Koopman observables above predict \emph{aggregate} resource usage, but do not directly inform \emph{how} to implement each frequency band (DSP-heavy multiplication vs.\ LUT-intensive dyadic quantization). We now connect the sensitivity matrix $\mathbf{S}$ from Chapter~3 (Eq.~\ref{eq:sensitivity_matrix}) to Koopman observables, enabling the MPC controller to request different implementation strategies based on predicted workload dynamics.

\paragraph{Sensitivity-Aware Resource Observables}

Recall from Chapter~3 that the sensitivity matrix $\mathbf{S} \in \mathbb{R}^{(N+1) \times 2}$ encodes how quantization errors in polynomial coefficients $c_n$ propagate to filter output error:
\[
S_n = |I_n| = \left| \int_{-\pi}^{\pi} x^{2n} R(x) \, dx \right|,
\]
where $R(x) = \widehat{H}(\omega) e^{-i\omega x}$ is the projection operator (Chapter~3, Section~\ref{sec:projector_derivation}). High-sensitivity coefficients ($S_n \gg \overline{S}$) require more fractional bits $m_n$ to maintain accuracy, directly increasing resource demands.

We augment the Koopman state vector with \emph{per-band sensitivity observables}:
\begin{align}
\psi_{\text{sens}}^{(j)}(\mathbf{x}) &= \max_{n \in \mathcal{N}_j} S_n^{(j)}, \quad \text{(peak sensitivity in band $j$)}, \label{eq:obs_sens_peak}\\
\psi_{\text{entropy}}^{(j)}(\mathbf{x}) &= -\sum_{n} \frac{S_n^{(j)}}{\|S^{(j)}\|_1} \log \frac{S_n^{(j)}}{\|S^{(j)}\|_1}, \quad \text{(sensitivity entropy)}, \label{eq:obs_sens_entropy}
\end{align}
where $S_n^{(j)}$ is the sensitivity of coefficient $n$ in Shannon wavelet scale $j$, and $\mathcal{N}_j = \{0, 1, \ldots, N_j\}$ indexes the polynomial terms for that band. These observables encode:
\begin{itemize}
    \item \textbf{Peak sensitivity} $\psi_{\text{sens}}^{(j)}$: Bands with high peak sensitivity require careful bit allocation; implementing them with \emph{dyadic quantization} (Chapter~5, Section~5.2.4) trades LUT usage for reduced DSP count via bit-shift arithmetic~\cite{Abdi2024,Gustafsson2007}.
    \item \textbf{Sensitivity entropy} $\psi_{\text{entropy}}^{(j)}$: Low entropy ($\psi_{\text{entropy}}^{(j)} \to 0$) indicates one or two coefficients dominate; these bands benefit from \emph{sparse polynomial evaluation} using DSP slices for the dominant terms only. High entropy ($\psi_{\text{entropy}}^{(j)} \to \log N_j$) suggests uniform importance across coefficients, favoring \emph{parallel LUT-based evaluation} of all terms~\cite{SinhaSrikanthan2014}.
\end{itemize}

\paragraph{Implementation Strategy Selection via Koopman Prediction}

The MPC controller uses predicted values ${\psi_{\text{sens}}^{(j)}}_{k+h|k}$, ${\psi_{\text{entropy}}^{(j)}}_{k+h|k}$ (evolved via $\mathbf{K}^h$) to select per-band implementation strategies:

\noindent\textbf{Example:} Suppose Koopman prediction indicates that at time $k+3$, frequency band $j=5$ will have ${\psi_{\text{sens}}^{(5)}}_{k+3|k} = 42.7 > \tau_{\text{high}}$ (high peak sensitivity) and ${\psi_{\text{entropy}}^{(5)}}_{k+3|k} = 0.8$ (low entropy, dominated by coefficients $c_2, c_3$). The MPC controller preemptively:
\begin{enumerate}
    \item Selects \emph{dyadic quantization mode} for band $j=5$ (Table~\ref{tab:sensitivity_impl_modes}, row 1).
    \item Invokes the BMT algorithm (Chapter~5, Section~5.2.4) or uses precomputed dyadic polynomial coefficients to decompose $c_2 \approx \frac{a_2}{2^{m_2}}$, $c_3 \approx \frac{a_3}{2^{m_3}}$ with $m_2, m_3 \geq 10$ fractional bits.
    \item Allocates LUT resources for bit-shift networks: $(x \ll m_2) + (x \ll (m_2-1)) + \cdots$ to evaluate $c_2 x^4 + c_3 x^6$ without DSP multipliers.
    \item Updates resource constraints: $\psi_{\text{DSP}}(\mathbf{x}_{k+3|k}) \gets \psi_{\text{DSP}}(\mathbf{x}_{k+3|k}) - 2$ (freed DSP slices from band 5), $\psi_{\text{LUT}}(\mathbf{x}_{k+3|k}) \gets \psi_{\text{LUT}}(\mathbf{x}_{k+3|k}) + 60$ (additional shift-add logic).
\end{enumerate}

\paragraph{Closed-Loop Refinement}

The learned coefficients $\alpha_j, \beta_j, \gamma_j$ in the resource observables can be refined online using the measured sensitivity matrix $\mathbf{S}_{\text{measured}}$ (computed from actual FPGA timing/resource reports after synthesis):
\[
\alpha_j^{\text{new}} = \alpha_j^{\text{old}} + \eta \left( \psi_{\text{DSP}}^{\text{actual}} - \psi_{\text{DSP}}^{\text{predicted}} \right) \frac{\partial \psi_{\text{DSP}}}{\partial \alpha_j},
\]
where $\eta$ is a learning rate. This creates a feedback mechanism (anticipated in Section~4.2.1, Eq.~\eqref{eq:dyadic_feedback_loop}): measured FPGA utilization $\to$ updates Koopman observable mappings $\to$ improves future resource predictions $\to$ refines sensitivity-driven mode selection.

\noindent\textbf{Integration with MPC Formulation}

The MPC optimization at time $k$ solves:
\[
\min_{\mathbf{m}^{(j)}_{k:k+H}} \sum_{h=0}^{H} \left\| \mathbf{e}_{k+h|k} \right\|^2 + \lambda \sum_{j,h} \left( m^{(j)}_{k+h} - m^{(j)}_{k+h-1} \right)^2
\]
subject to:
\begin{align}
\mathbf{\Psi}_{k+h+1|k} &= \mathbf{K} \mathbf{\Psi}_{k+h|k}, \\
\sum_{j} \psi_{\text{DSP}}(\mathbf{x}_{k+h|k}) &\leq C_{\text{DSP}}, \\
\sum_{j} \psi_{\text{LUT}}(\mathbf{x}_{k+h|k}) &\leq C_{\text{LUT}}, \\
m^{(j)}_{\min} \leq m^{(j)}_{k+h} &\leq m^{(j)}_{\max},
\end{align}
where $\mathbf{e}_{k+h|k}$ is the predicted quantization error (from Chapter~3, Eq.~3.15), $\mathbf{m}^{(j)}$ is the bit allocation vector for Shannon wavelet scale $j$, and $C_{\text{DSP}}, C_{\text{LUT}}$ are FPGA resource limits. The regularization term $\lambda(\cdot)^2$ penalizes rapid reconfigurations to avoid excessive partial reconfiguration overhead~\cite{Bhandari2012}.

\noindent\textbf{Key advantage:} Explicit MPC for embedded systems~\cite{Bemporad2002} precomputes control laws offline, enabling $O(1)$ runtime lookup. Combined with sparse Koopman identification~\cite{Kaiser2018}, this yields real-time feasibility on FPGA fabric.
